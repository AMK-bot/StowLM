def sha(path):

def audit_target(value,dir_fd=-1):
 path=Path(os.fsdecode(value))
 if path.is_absolute():return path
 if isinstance(dir_fd,int) and dir_fd>=0:
  if sys.platform=='darwin':
   import fcntl
   base=fcntl.fcntl(dir_fd,getattr(fcntl,'F_GETPATH',50),b'\0'*1024).split(b'\0',1)[0]
   return Path(os.fsdecode(base))/path
  return Path(os.readlink(f'/proc/self/fd/{dir_fd}'))/path
 return path.absolute()

def guard_repository_writes(row):
 fd=os.open(WORK/f"writes-{row['seed']}.jsonl",os.O_WRONLY|os.O_CREAT|os.O_APPEND,0o600)
 def hook(event,args):
  paths=[]
  if event=='open':
   path,mode,flags=args
   if not ((isinstance(mode,str) and any(c in mode for c in 'wax+')) or isinstance(flags,int) and flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC)):return
   paths=[(path,-1)]
  elif event=='os.rename':paths=[(args[0],args[2]),(args[1],args[3])]
  elif event in ('os.remove','os.rmdir'):paths=[(args[0],args[1])]
  elif event=='os.mkdir':paths=[(args[0],args[2])]
  else:return
  for value,dir_fd in paths:
   if not isinstance(value,(str,bytes,os.PathLike)):continue
   path=audit_target(value,dir_fd)
   if path.is_relative_to(ROOT):
    allowed=path.is_relative_to(HOME)
    os.write(fd,(json.dumps(dict(event=event,path=str(path.relative_to(ROOT)),dir_fd=dir_fd,allowed=allowed,time=time.time()))+'\n').encode())
    if not allowed:raise RuntimeError('Extension generation attempted a repository write outside its excluded fence')
 sys.addaudithook(hook)
