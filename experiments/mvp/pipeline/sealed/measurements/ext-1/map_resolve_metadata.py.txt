"""Close a point-in-time metadata census across unrelated recording refreshes."""
import hashlib,importlib.machinery,importlib.util,json,os,time
from pathlib import Path
HOME=Path(__file__).resolve().parent;ROOT=HOME.parents[5]
def load(name,file):
 loader=importlib.machinery.SourceFileLoader(name,str(HOME/file));spec=importlib.util.spec_from_loader(name,loader);m=importlib.util.module_from_spec(spec);loader.exec_module(m);return m
audit=load('ext1_audit_resolve','map_seed_audit.py.txt');complete=load('ext1_complete_resolve','map_metadata_complete.py.txt')
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def atomic(path,value):
 tmp=path.with_name(path.name+'.resolved.tmp');tmp.write_text(json.dumps(value,indent=2)+'\n');os.replace(tmp,path)
def main():
 proof_path=HOME/'map-seed-proof.json';inventory_path=HOME/'map-prior-identities.json'
 proof=json.loads(proof_path.read_text());inventory=json.loads(inventory_path.read_text());approved='experiments/mvp/video/site/play/rec/index.json'
 errors=[row for row in proof['errors'] if not (row['path']==approved and row['error']=='metadata_changed_since_census')]
 refreshes=[];checked=0
 for entry in proof['files']:
  path=ROOT/entry['path'];raw=path.read_bytes();digest=hashlib.sha256(raw).hexdigest();checked+=1
  if digest==entry['sha256']:continue
  if entry['path']!=approved:
   errors.append(dict(path=entry['path'],error='identity_metadata_changed_at_final_cutoff'));continue
  text=raw.decode('utf-8');value=json.loads(text,object_pairs_hook=complete.Pairs);seeds,ranges=complete.seed_records(value)
  if ranges:errors.append(dict(path=approved,error='recording_index_unexpected_seed_ranges'));continue
  hits=audit.proposed_hits(text)
  if hits:proof['collisions'].append(dict(path=approved,source='refreshed_public_recording_index',seeds=hits))
  for seed in seeds:
   key=str(seed);inventory['seeds'].setdefault(key,[])
   if approved not in inventory['seeds'][key]:inventory['seeds'][key].append(approved)
  for match in audit.HEX.finditer(text):
   key=match.group().lower();inventory['digests'].setdefault(key,[])
   if approved not in inventory['digests'][key]:inventory['digests'][key].append(approved)
  refreshes.append(dict(path=approved,previous_observed_sha256=entry['sha256'],final_observed_sha256=digest,
   identity_merge='Union of every 64-hex value from both observed versions; old inventory retained',
   classification='Non-cohort HTTP recording lookup metadata, generated by ui/static_play/build.py:439-441; no seed interval fields',
   producer='experiments/mvp/ui/static_play/build.py',producer_sha256=sha(ROOT/'experiments/mvp/ui/static_play/build.py')))
  entry.update(sha256=digest,bytes=len(raw))
 atomic(inventory_path,inventory)
 proof.update(errors=errors,metadata_cutoff_utc=time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),
  final_metadata_hashes_rechecked=checked,concurrent_non_cohort_metadata_refreshes=refreshes,
  structural_seed_coverage_complete=not errors,no_seed_collision=not errors and not proof['collisions'],
  ready_for_generation=not errors and not proof['collisions'],prior_digest_count=len(inventory['digests']),
  prior_identities_sha256=sha(inventory_path),
  metadata_inventory_sha256=hashlib.sha256(json.dumps(proof['files'],sort_keys=True,separators=(',',':')).encode()).hexdigest(),
  prior_seed_count_definition='Conservative recorded seed-tagged integer/path values, not number of prior maps',
  census_time_limit='Recorded external identities as of the stated cutoff. Unrecorded/future identities and forbidden raw payloads are not claimed inspected.')
 atomic(proof_path,proof)
 print(json.dumps({key:proof[key] for key in ('ready_for_generation','prior_seed_count','prior_digest_count','metadata_files_checked','structural_json_files','collisions','errors','concurrent_non_cohort_metadata_refreshes')}))
 return 0 if proof['ready_for_generation'] else 2
if __name__=='__main__':raise SystemExit(main())
