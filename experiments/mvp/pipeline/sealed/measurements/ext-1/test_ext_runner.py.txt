def save_config(setup, **changes):
    setup['config'].update(changes)
    atomic_json(setup['path'], setup['config'])
    setup['effective'] = subject.load_config(setup['path'])


def process_files(target, request, *, phase='plan', elapsed=1, returncode=0, timeout=False, cleanup=False, pid=None):
    audit = phase == 'audit'
    prefix = 'audit' if audit else 'child'
    pid = pid or (456 if audit else 123)
    atomic_json(target/('audit-job.json' if audit else 'job.json'), request)
    deadline = 60 if phase in ('audit','prepass') else 20
    atomic_json(target/('audit.pid.json' if audit else 'pid.json'),
        dict(pid=pid,pgid=pid,start_new_session=True,started=time.time()-elapsed,timeout_s=deadline))
    metrics = {prefix+'_elapsed_s':elapsed,prefix+'_returncode':returncode,
               prefix+'_timed_out':timeout,prefix+'_cleanup_failed':cleanup,prefix+'_pid':pid,
               prefix+'_timeout_s':deadline}
    atomic_json(target/('audit.process.json' if audit else 'process.json'), metrics)
    return metrics


def fake_execution(calls):
    def execute(config, phase, request, target):
        calls.append(phase)
        metrics = process_files(target, request, phase=phase)
        if phase == 'plan':
            row = dict(map_id=request['map_id'],map_sha256=request['sha256'],cohort=request['cohort'],
                arm=request['arm'],status='checked_plan',level='L0',passed=True,full_checker_pass=True,
                planning_pid=123,plan={'moves':[]},audit_verified=False, audit_fresh_process=False)
            atomic_json(target/'row.json', row)
        elif phase == 'audit':
            row = dict(json.loads(Path(request['target']).read_text()), audit_verified=True,
                       audit_fresh_process=True,audit_pid=456,audited_target_sha256=request['target_sha256'])
            atomic_json(target/'audit-row.json', row)
        else:
            raise AssertionError(phase)
        return dict(row, **metrics)
    return execute

@pytest.mark.parametrize('event', ['os.remove','os.rmdir','os.mkdir','os.chmod','os.utime'])
def test_native_fd_relative_cleanup_resolves_owned_directory(setup,event):
    fd=os.open(setup['home'],os.O_RDONLY)
    try:
        args={'os.remove':('relative.json',fd),'os.rmdir':('relative-dir',fd),
              'os.mkdir':('relative-dir',0o777,fd),'os.chmod':('relative.json',0o600,fd),
              'os.utime':('relative.json',None,None,fd)}[event]
        subject.repository_write_hook(event,args)
        assert subject.audit_target('relative.json',fd) == setup['home']/'relative.json'
    finally:
        os.close(fd)

def test_rename_resolves_both_source_and_destination_directory_fds(setup):
    directory=setup['root']/'experiments/mvp/frontier/engine'

    own=os.open(setup['home'],os.O_RDONLY)

@pytest.mark.parametrize('with_incumbent',[False,True])
def test_completed_deadline_audit_is_preserved_after_crash_before_result(setup,monkeypatch,with_incumbent):
    target=target_for(setup)
    config=setup['effective']
    request=subject._issue(config,subject._base_job(config,setup['item'],'pipeline'),'plan','plan-once')
    metrics=process_files(target,request,elapsed=20,timeout=True,returncode=-9)
    observed={}
    if with_incumbent:
        plan={'moves':[]}
        observed[object_hash(plan)]=dict(plan=plan,lifts=1,published_wall_s=10,
            publication_sha256='b'*64,validation_source='runner_checked_incumbent')
    raw=driver._deadline_result(request,target,metrics,observed)

    audit=subject._issue(config,dict(subject._base_job(config,setup['item'],'pipeline'),
        target=str(target/'deadline-row.json'),target_sha256=sha256(target/'deadline-row.json'),
        deadline_recovery=True),'audit','audit-once')

    row.update(passed=with_incumbent,full_checker_pass=with_incumbent,level='L0' if with_incumbent else None,
               deadline_with_incumbent=with_incumbent)
